    36→    36→/* discretize a 0..1 value into N zones, returns zone index 0..N-1 */
    37→    37→function zone(v: number, n: number) {
    38→    38→  return Math.max(0, Math.min(n - 1, Math.floor(v * n)));
    39→    39→}

    41→    41→export function OrchestraView() {
    42→    42→  const canvasRef = useRef<HTMLCanvasElement>(null);
    43→    43→  const engineRef = useRef<MusicEngine | null>(null);
    44→    44→  const lockedRef = useRef(false);
    45→    45→  const prevRightGestureRef = useRef<GestureType>("idle");
    46→    46→  const prevLeftGestureRef = useRef<GestureType>("idle");
    47→    47→  const videoElRef = useRef<HTMLVideoElement | null>(null);
    48→    48→
    49→    49→  const [running, setRunning] = useState(false);
    50→    50→  const [loading, setLoading] = useState(false);
    51→    51→  const [error, setError] = useState<string | null>(null);
    52→    52→  const [rightGesture, setRightGesture] = useState<GestureType>("idle");
    53→    53→  const [leftGesture, setLeftGesture] = useState<GestureType>("idle");
    54→    54→  const [rightPresent, setRightPresent] = useState(false);
    55→    55→  const [leftPresent, setLeftPresent] = useState(false);
    56→    56→  const [tempo, setTempo] = useState(90);
    57→    57→  const [dynamicLevel, setDynamicLevel] = useState(3);
    58→    58→  const [chordName, setChordName] = useState("I");
    59→    59→  const loadPref = <T,>(key: string, fallback: T): T => {
    60→    60→    if (typeof window === "undefined") return fallback;
    61→    61→    try { const v = localStorage.getItem(key); return v === null ? fallback : JSON.parse(v) as T; } catch { return fallback; }
    62→    62→  };
    63→    63→  const [scaleIdx, setScaleIdx] = useState(() => loadPref("orch:scale", 0));
    64→    64→  const [progIdx, setProgIdx] = useState(() => loadPref("orch:prog", 0));
    65→    65→  const [melodyNote, setMelodyNote] = useState("C5");
    66→    66→  const [melodyDegree, setMelodyDegree] = useState(0);
    67→    67→  const [melodyOn, setMelodyOn] = useState(true);
    68→    68→  const [locked, setLocked] = useState(false);
    69→    69→  const [activeLayers, setActiveLayers] = useState<LayerName[]>(["strings", "piano", "bass", "drums", "lead"]);
    70→    70→  const [layerVols, setLayerVols] = useState<Record<LayerName, number>>(() => loadPref("orch:vols", {
    71→    71→    strings: 0.5, piano: 0.6, bass: 0.7, drums: 0.7, lead: 0.55,
    72→    72→  }));
    73→    73→  const [reverbAmt, setReverbAmt] = useState(() => loadPref("orch:reverb", 0.3));
    74→    74→  const [swingAmt, setSwingAmt] = useState(() => loadPref("orch:swing", 0));
    75→    75→  const [octave, setOctave] = useState(() => loadPref("orch:octave", 0));
    76→    76→  const [showMix, setShowMix] = useState(false);
    77→    77→  const [rightX, setRightX] = useState(0.5);
    78→    78→  const [leftX, setLeftX] = useState(0.5);
    79→    79→  const [leftY, setLeftY] = useState(0.5);
    80→    80→  const [rightY, setRightY] = useState(0.5);
    81→    81→
    82→    82→  // persist prefs to localStorage
    83→    83→  useEffect(() => { localStorage.setItem("orch:scale", JSON.stringify(scaleIdx)); }, [scaleIdx]);
    84→    84→  useEffect(() => { localStorage.setItem("orch:prog", JSON.stringify(progIdx)); }, [progIdx]);
    85→    85→  useEffect(() => { localStorage.setItem("orch:vols", JSON.stringify(layerVols)); }, [layerVols]);
    86→    86→  useEffect(() => { localStorage.setItem("orch:reverb", JSON.stringify(reverbAmt)); }, [reverbAmt]);
    87→    87→  useEffect(() => { localStorage.setItem("orch:swing", JSON.stringify(swingAmt)); }, [swingAmt]);
    88→    88→  useEffect(() => { localStorage.setItem("orch:octave", JSON.stringify(octave)); }, [octave]);
    89→    89→
    90→    90→  /* -------------------------------------------------------------- */
    91→    91→  /*  Per-frame: draw skeletons + drive engine from TWO hands        */
    92→    92→  /* -------------------------------------------------------------- */
    93→    93→  const onHands = useCallback((hands: HandState[], lms: (Landmark[] | null)[]) => {
    94→    94→    const canvas = canvasRef.current;
    95→    95→    const video = videoElRef.current;
    96→    96→    const eng = engineRef.current;
    97→    97→
    98→    98→    // identify right & left hands
    99→    99→    let right: HandState | null = null;
   100→   100→    let left: HandState | null = null;
   101→   101→    let rightLm: Landmark[] | null = null;
   102→   102→    let leftLm: Landmark[] | null = null;
   103→   103→    for (let i = 0; i < hands.length; i++) {
   104→   104→      const h = hands[i];
   105→   105→      if (!h.present) continue;
   106→   106→      if (h.handedness === "Right" && !right) { right = h; rightLm = lms[i]; }
   107→   107→      else if (h.handedness === "Left" && !left) { left = h; leftLm = lms[i]; }
   108→   108→      else if (!right) { right = h; rightLm = lms[i]; }
   109→   109→      else if (!left) { left = h; leftLm = lms[i]; }
   110→   110→    }
   111→   111→
   112→   112→    setRightPresent(!!right?.present);
   113→   113→    setLeftPresent(!!left?.present);
   114→   114→
   115→   115→    // draw skeletons
   116→   116→    if (canvas && video) {
   117→   117→      const ctx = canvas.getContext("2d")!;
   118→   118→      canvas.width = video.videoWidth || 640;
   119→   119→      canvas.height = video.videoHeight || 480;
   120→   120→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   121→   121→      const drawHand = (lm: Landmark[] | null, color: string) => {
   122→   122→        if (!lm) return;
   123→   123→        const w = canvas!.width, hh = canvas!.height;
   124→   124→        ctx.strokeStyle = color;
   125→   125→        ctx.lineWidth = 3;
   126→   126→        ctx.lineCap = "round";
   127→   127→        for (const [a, b] of HAND_CONNECTIONS) {
   128→   128→          ctx.beginPath();
   129→   129→          ctx.moveTo(lm[a].x * w, lm[a].y * hh);
   130→   130→          ctx.lineTo(lm[b].x * w, lm[b].y * hh);
   131→   131→          ctx.stroke();
   132→   132→        }
   133→   133→        ctx.fillStyle = "rgba(255,210,80,0.95)";
   134→   134→        for (const p of lm) {
   135→   135→          ctx.beginPath();
   136→   136→          ctx.arc(p.x * w, p.y * hh, 5, 0, Math.PI * 2);
   137→   137→          ctx.fill();
   138→   138→        }
   139→   139→      };
   140→   140→      drawHand(rightLm, "rgba(0,255,140,0.9)");
   141→   141→      drawHand(leftLm, "rgba(255,120,80,0.9)");
   142→   142→    }
   143→   143→
   144→   144→    if (!eng) return;
   145→   145→
   146→   146→    /* ---- RIGHT HAND: tempo + melody ---- */
   147→   147→    if (right) {
   148→   148→      setRightGesture(right.gesture);
   149→   149→      setRightX(right.x);
   150→   150→      setRightY(right.y);
   151→   151→      // Y -> tempo (top = fast). 50..180
   152→   152→      const bpm = 50 + (1 - right.y) * 130;
   153→   153→      eng.setTempo(bpm);
   154→   154→      setTempo(Math.round(bpm));
   155→   155→      // X -> melody degree (7 zones)
   156→   156→      const n = SCALES[scaleIdx].intervals.length;
   157→   157→      const deg = zone(right.x, n * 2);
   158→   158→      eng.setMelodyDegree(deg);
   159→   159→      setMelodyDegree(deg);
   160→   160→
   161→   161→      // gesture edges
   162→   162→      const g = right.gesture;
   163→   163→      const prev = prevRightGestureRef.current;
   164→   164→      if (g !== prev) {
   165→   165→        if (g === "fist") { eng.setMelodyEnabled(false); setMelodyOn(false); }
   166→   166→        if (g === "open") { eng.setMelodyEnabled(true); setMelodyOn(true); }
   167→   167→        if (g === "point") {
   168→   168→          // accent: re-trigger melody note now
   169→   169→          eng.setMelodyEnabled(true); setMelodyOn(true);
   170→   170→        }
   171→   171→        prevRightGestureRef.current = g;
   172→   172→      }
   173→   173→    } else {
   174→   174→      setRightGesture("idle");
   175→   175→    }
   176→   176→
   177→   177→    /* ---- LEFT HAND: dynamics + progression ---- */
   178→   178→    if (left) {
   179→   179→      setLeftGesture(left.gesture);
   180→   180→      setLeftX(left.x);
   181→   181→      setLeftY(left.y);
   182→   182→      // Y -> dynamic level (6 zones, top = ff)
   183→   183→      const dl = 5 - zone(left.y, 6);
   184→   184→      eng.setDynamicLevel(dl);
   185→   185→      setDynamicLevel(dl);
   186→   186→      // X -> progression (4 zones)
   187→   187→      const pi = zone(left.x, PROGRESSIONS.length);
   188→   188→      if (pi !== progIdx) {
   189→   189→        eng.setProgression(pi);
   190→   190→        setProgIdx(pi);
   191→   191→      }
   192→   192→
   193→   193→      // gesture edges
   194→   194→      const g = left.gesture;
   195→   195→      const prev = prevLeftGestureRef.current;
   196→   196→      if (g !== prev) {
   197→   197→        if (g === "open") {
   198→   198→          // full ensemble
   199→   199→          (["strings", "piano", "bass", "drums", "lead"] as LayerName[]).forEach((l) => {
   200→   200→            eng.setLayerVolume(l, layerVols[l] || 0.6);
   201→   201→          });
   202→   202→          setActiveLayers((Object.keys(layerVols) as LayerName[]).filter((l) => (layerVols[l] || 0) > 0.01));
   203→   203→        }
   204→   204→        if (g === "fist") {
   205→   205→          // strip to strings only
   206→   206→          (["piano", "bass", "drums", "lead"] as LayerName[]).forEach((l) => eng.setLayerVolume(l, 0));
   207→   207→          eng.setLayerVolume("strings", layerVols.strings || 0.6);
   208→   208→          setActiveLayers(["strings"]);
   209→   209→        }
   210→   210→        if (g === "pinch") {
   211→   211→          lockedRef.current = !lockedRef.current;
   212→   212→          eng.setLocked(lockedRef.current);
   213→   213→          setLocked(lockedRef.current);
   214→   214→        }
   215→   215→        prevLeftGestureRef.current = g;
   216→   216→      }
   217→   217→    } else {
   218→   218→      setLeftGesture("idle");
   219→   219→    }
   220→   220→
   221→   221→    // sync display state from engine
   222→   222→    if (eng.state) {
   223→   223→      setChordName(eng.state.currentChordName);
   224→   224→      setMelodyNote(eng.state.melodyNote);
   225→   225→      setActiveLayers([...eng.state.activeLayers]);
   226→   226→    }
   227→   227→  }, [scaleIdx, progIdx, layerVols]);
   228→   228→
   229→   229→  const tracking = useHandTracking({ onHands, numHands: 2, smoothing: 0.45 });
   230→   230→  useEffect(() => {
   231→   231→    videoElRef.current = tracking.videoRef.current;
   232→   232→  });
   233→   233→
   234→   234→  /* -------------------------------------------------------------- */
   235→   235→  /*  Start / stop                                                  */
   236→   236→  /* -------------------------------------------------------------- */
   237→   237→  const start = async () => {
   238→   238→    setError(null);
   239→   239→    setLoading(true);
   240→   240→    try {
   241→   241→      if (!engineRef.current) {
   242→   242→        engineRef.current = new MusicEngine();
   243→   243→        await engineRef.current.init();
   244→   244→      }
   245→   245→      // apply current UI state
   246→   246→      engineRef.current.setScale(scaleIdx);
   247→   247→      engineRef.current.setProgression(progIdx);
   248→   248→      engineRef.current.setReverb(reverbAmt);
   249→   249→      engineRef.current.setSwing(swingAmt);
   250→   250→      engineRef.current.setOctave(octave);
   251→   251→      engineRef.current.setDynamicLevel(dynamicLevel);
   252→   252→      (Object.keys(layerVols) as LayerName[]).forEach((l) => engineRef.current!.setLayerVolume(l, layerVols[l]));
   253→   253→      engineRef.current.start();
   254→   254→      await tracking.start();
   255→   255→      setRunning(true);
   256→   256→    } catch (e) {
   257→   257→      setError(e instanceof Error ? e.message : "Failed to start");
   258→   258→    } finally {
   259→   259→      setLoading(false);
   260→   260→    }
   261→   261→  };
   262→   262→
   263→   263→  const stop = () => {
   264→   264→    tracking.stop();
   265→   265→    setRunning(false);
   266→   266→    engineRef.current?.stop();
   267→   267→    setRightGesture("idle");
   268→   268→    setLeftGesture("idle");
   269→   269→    setRightPresent(false);
   270→   270→    setLeftPresent(false);
   271→   271→  };
   272→   272→
   273→   273→  useEffect(() => {
   274→   274→    return () => {
   275→   275→      engineRef.current?.dispose();
   276→   276→      engineRef.current = null;
   277→   277→    };
   278→   278→  }, []);
   279→   279→
   280→   280→  /* ---- UI handlers ---- */
   281→   281→  const onScale = (i: number) => { setScaleIdx(i); engineRef.current?.setScale(i); };
   282→   282→  const onProg = (i: number) => { setProgIdx(i); engineRef.current?.setProgression(i); };
   283→   283→  const onLayerVol = (l: LayerName, v: number) => {
   284→   284→    setLayerVols((s) => ({ ...s, [l]: v }));
   285→   285→    engineRef.current?.setLayerVolume(l, v);
   286→   286→  };
   287→   287→  const onReverb = (v: number) => { setReverbAmt(v); engineRef.current?.setReverb(v); };
   288→   288→  const onSwing = (v: number) => { setSwingAmt(v); engineRef.current?.setSwing(v); };
   289→   289→  const onOctave = (v: number) => { setOctave(v); engineRef.current?.setOctave(v); };
   290→   290→  const onDrop = () => engineRef.current?.triggerDrop();
   291→   291→  const onPanic = () => {
   292→   292→    engineRef.current?.stop();
   293→   293→    setTimeout(() => engineRef.current?.start(), 100);
   294→   294→  };
   295→   295→
   296→   296→  const intensity = dynamicLevel / 5;
   297→   297→
   298→   298→  return (
   299→   299→    <div className="relative min-h-screen pt-16 sm:pt-20 pb-12">
   300→   300→      {/* ambient background */}
   301→   301→      <div className="absolute inset-0 overflow-hidden pointer-events-none">
   302→   302→        <motion.div
   303→   303→          className="absolute top-1/4 left-1/4 h-96 w-96 rounded-full blur-3xl"
   304→   304→          style={{ background: `radial-gradient(circle, oklch(0.7 0.2 305 / ${0.15 + intensity * 0.25}), transparent)` }}
   305→   305→          animate={{ scale: 1 + intensity * 0.3 }}
   306→   306→          transition={{ duration: 0.4 }}
   307→   307→        />
   308→   308→        <motion.div
   309→   309→          className="absolute bottom-1/4 right-1/4 h-96 w-96 rounded-full blur-3xl"
   310→   310→          style={{ background: `radial-gradient(circle, oklch(0.72 0.19 155 / ${0.1 + intensity * 0.2}), transparent)` }}
   311→   311→          animate={{ scale: 1 + intensity * 0.3 }}
   312→   312→          transition={{ duration: 0.4 }}
   313→   313→        />
   314→   314→      </div>
   315→   315→
   316→   316→      <div className="relative mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   317→   317→        {/* header */}
   318→   318→        <div className="mb-8 text-center">
   319→   319→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="inline-flex items-center gap-2 px-3 py-1 rounded-full glass text-xs font-medium text-chart-4 mb-4">
   320→   320→            <Music className="h-3.5 w-3.5" /> Two-handed conducting · Generative · Tone.js
   321→   321→          </motion.div>
   322→   322→          <h1 className="text-4xl sm:text-5xl font-bold tracking-tight">
   323→   323→            AI Conducting <span className="text-gradient">Orchestra</span>
   324→   324→          </h1>
   325→   325→          <p className="mt-3 text-muted-foreground max-w-2xl mx-auto">
   326→   326→            Conduct a live, procedurally-generated cinematic score with <span className="text-primary">both hands</span>.
   327→   327→            Right hand sets tempo &amp; melody; left hand sets dynamics &amp; harmony. Every note is synthesized in real time.
   328→   328→          </p>
   329→   329→        </div>
   330→   330→
   331→   331→        <div className="grid lg:grid-cols-[1.3fr_1fr] gap-6">
   332→   332→          {/* ---- camera + overlays ---- */}
   333→   333→          <div className="flex flex-col gap-4">
   334→   334→            <div className="relative rounded-2xl overflow-hidden glass-strong aspect-video bg-black">
   335→   335→              <video ref={tracking.videoRef} playsInline muted className="absolute inset-0 h-full w-full object-cover opacity-0" />
   336→   336→              <canvas ref={canvasRef} className="absolute inset-0 h-full w-full object-cover" />
   337→   337→
   338→   338→              {!running && (
   339→   339→                <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 grid-bg">
   340→   340→                  <div className="relative">
   341→   341→                    <div className="absolute inset-0 bg-chart-4/30 blur-2xl rounded-full animate-pulse-glow" />
   342→   342→                    <Music className="relative h-16 w-16 text-chart-4/70" />
   343→   343→                  </div>
   344→   344→                  <p className="text-muted-foreground text-sm max-w-xs text-center">
   345→   345→                    {loading ? "Loading model + audio engine…" : "Click start to begin conducting"}
   346→   346→                  </p>
   347→   347→                  {loading && <Loader2 className="h-5 w-5 animate-spin text-chart-4" />}
   348→   348→                </div>
   349→   349→              )}
   350→   350→
   351→   351→              <AnimatePresence>
   352→   352→                {running && (
   353→   353→                  <>
   354→   354→                    {/* right hand badge */}
   355→   355→                    <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="absolute top-3 left-3 flex items-center gap-2 px-3 py-1.5 rounded-lg glass-strong">
   356→   356→                      <span className={cn("h-2 w-2 rounded-full", rightPresent ? "bg-primary animate-pulse" : "bg-muted-foreground")} />
   357→   357→                      <span className="text-xs font-medium">R: <span className="capitalize text-primary">{rightGesture}</span></span>
   358→   358→                    </motion.div>
   359→   359→                    {/* left hand badge */}
   360→   360→                    <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="absolute top-3 right-3 flex items-center gap-2 px-3 py-1.5 rounded-lg glass-strong">
   361→   361→                      <span className={cn("h-2 w-2 rounded-full", leftPresent ? "bg-chart-3 animate-pulse" : "bg-muted-foreground")} />
   362→   362→                      <span className="text-xs font-medium">L: <span className="capitalize text-chart-3">{leftGesture}</span></span>
   363→   363→                    </motion.div>
   364→   364→                    {/* chord + melody */}
   365→   365→                    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="absolute bottom-3 left-3 flex gap-2">
   366→   366→                      <div className="px-3 py-1.5 rounded-lg glass-strong text-center">
   367→   367→                        <div className="text-[9px] text-muted-foreground uppercase">Chord</div>
   368→   368→                        <div className="text-lg font-bold text-chart-3 leading-none">{chordName}</div>
   369→   369→                      </div>
   370→   370→                      <div className="px-3 py-1.5 rounded-lg glass-strong text-center">
   371→   371→                        <div className="text-[9px] text-muted-foreground uppercase">Melody</div>
   372→   372→                        <div className="text-lg font-bold text-chart-4 leading-none">{melodyNote}</div>
   373→   373→                      </div>
   374→   374→                    </motion.div>
   375→   375→                    {locked && (
   376→   376→                      <motion.div initial={{ opacity: 0, scale: 0.8 }} animate={{ opacity: 1, scale: 1 }} className="absolute bottom-3 right-3 flex items-center gap-1.5 px-3 py-1.5 rounded-lg glass-strong">
   377→   377→                        <Lock className="h-3.5 w-3.5 text-chart-3" />
   378→   378→                        <span className="text-xs">Locked</span>
   379→   379→                      </motion.div>
   380→   380→                    )}
   381→   381→                  </>
   382→   382→                )}
