     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import {
     6→  Camera, CameraOff, Loader2, Music, Hand, Volume2, Activity,
     7→  Play, Pause, Lock, Zap, RotateCcw, Sliders, ChevronRight,
     8→} from "lucide-react";
     9→import { cn } from "@/lib/utils";
    10→import {
    11→  useHandTracking, type HandState, type Landmark, type GestureType,
    12→} from "@/lib/gesture/use-hand-tracking";
    13→import {
    14→  MusicEngine, type LayerName, SCALES, PROGRESSIONS,
    15→} from "@/lib/gesture/music-engine";
    16→
    17→const HAND_CONNECTIONS: [number, number][] = [
    18→  [0, 1], [1, 2], [2, 3], [3, 4],
    19→  [0, 5], [5, 6], [6, 7], [7, 8],
    20→  [5, 9], [9, 10], [10, 11], [11, 12],
    21→  [9, 13], [13, 14], [14, 15], [15, 16],
    22→  [13, 17], [17, 18], [18, 19], [19, 20],
    23→  [0, 17],
    24→];
    25→
    26→const LAYERS: { id: LayerName; label: string; color: string }[] = [
    27→  { id: "strings", label: "Strings", color: "text-primary" },
    28→  { id: "piano",   label: "Piano",   color: "text-chart-2" },
    29→  { id: "bass",    label: "Bass",    color: "text-chart-3" },
    30→  { id: "drums",   label: "Drums",   color: "text-chart-5" },
    31→  { id: "lead",    label: "Lead",    color: "text-chart-4" },
    32→];
    33→
    34→const DYN_LABELS = ["pp", "p", "mp", "mf", "f", "ff"];
    35→
    36→/* discretize a 0..1 value into N zones, returns zone index 0..N-1 */
    37→function zone(v: number, n: number) {
    38→  return Math.max(0, Math.min(n - 1, Math.floor(v * n)));
    39→}
    40→
    41→export function OrchestraView() {
    42→  const canvasRef = useRef<HTMLCanvasElement>(null);
    43→  const engineRef = useRef<MusicEngine | null>(null);
    44→  const lockedRef = useRef(false);
    45→  const prevRightGestureRef = useRef<GestureType>("idle");
    46→  const prevLeftGestureRef = useRef<GestureType>("idle");
    47→  const videoElRef = useRef<HTMLVideoElement | null>(null);
    48→
    49→  const [running, setRunning] = useState(false);
    50→  const [loading, setLoading] = useState(false);
    51→  const [error, setError] = useState<string | null>(null);
    52→  const [rightGesture, setRightGesture] = useState<GestureType>("idle");
    53→  const [leftGesture, setLeftGesture] = useState<GestureType>("idle");
    54→  const [rightPresent, setRightPresent] = useState(false);
    55→  const [leftPresent, setLeftPresent] = useState(false);
    56→  const [tempo, setTempo] = useState(90);
    57→  const [dynamicLevel, setDynamicLevel] = useState(3);
    58→  const [chordName, setChordName] = useState("I");
    59→  const loadPref = <T,>(key: string, fallback: T): T => {
    60→    if (typeof window === "undefined") return fallback;
    61→    try { const v = localStorage.getItem(key); return v === null ? fallback : JSON.parse(v) as T; } catch { return fallback; }
    62→  };
    63→  const [scaleIdx, setScaleIdx] = useState(() => loadPref("orch:scale", 0));
    64→  const [progIdx, setProgIdx] = useState(() => loadPref("orch:prog", 0));
    65→  const [melodyNote, setMelodyNote] = useState("C5");
    66→  const [melodyDegree, setMelodyDegree] = useState(0);
    67→  const [melodyOn, setMelodyOn] = useState(true);
    68→  const [locked, setLocked] = useState(false);
    69→  const [activeLayers, setActiveLayers] = useState<LayerName[]>(["strings", "piano", "bass", "drums", "lead"]);
    70→  const [layerVols, setLayerVols] = useState<Record<LayerName, number>>(() => loadPref("orch:vols", {
    71→    strings: 0.5, piano: 0.6, bass: 0.7, drums: 0.7, lead: 0.55,
    72→  }));
    73→  const [reverbAmt, setReverbAmt] = useState(() => loadPref("orch:reverb", 0.3));
    74→  const [swingAmt, setSwingAmt] = useState(() => loadPref("orch:swing", 0));
    75→  const [octave, setOctave] = useState(() => loadPref("orch:octave", 0));
    76→  const [showMix, setShowMix] = useState(false);
    77→  const [rightX, setRightX] = useState(0.5);
    78→  const [leftX, setLeftX] = useState(0.5);
    79→  const [leftY, setLeftY] = useState(0.5);
    80→  const [rightY, setRightY] = useState(0.5);
    81→
    82→  // persist prefs to localStorage
    83→  useEffect(() => { localStorage.setItem("orch:scale", JSON.stringify(scaleIdx)); }, [scaleIdx]);
    84→  useEffect(() => { localStorage.setItem("orch:prog", JSON.stringify(progIdx)); }, [progIdx]);
    85→  useEffect(() => { localStorage.setItem("orch:vols", JSON.stringify(layerVols)); }, [layerVols]);
    86→  useEffect(() => { localStorage.setItem("orch:reverb", JSON.stringify(reverbAmt)); }, [reverbAmt]);
    87→  useEffect(() => { localStorage.setItem("orch:swing", JSON.stringify(swingAmt)); }, [swingAmt]);
    88→  useEffect(() => { localStorage.setItem("orch:octave", JSON.stringify(octave)); }, [octave]);
    89→
    90→  /* -------------------------------------------------------------- */
    91→  /*  Per-frame: draw skeletons + drive engine from TWO hands        */
    92→  /* -------------------------------------------------------------- */
    93→  const onHands = useCallback((hands: HandState[], lms: (Landmark[] | null)[]) => {
    94→    const canvas = canvasRef.current;
    95→    const video = videoElRef.current;
    96→    const eng = engineRef.current;
    97→
    98→    // identify right & left hands
    99→    let right: HandState | null = null;
   100→    let left: HandState | null = null;
   101→    let rightLm: Landmark[] | null = null;
   102→    let leftLm: Landmark[] | null = null;
   103→    for (let i = 0; i < hands.length; i++) {
   104→      const h = hands[i];
   105→      if (!h.present) continue;
   106→      if (h.handedness === "Right" && !right) { right = h; rightLm = lms[i]; }
   107→      else if (h.handedness === "Left" && !left) { left = h; leftLm = lms[i]; }
   108→      else if (!right) { right = h; rightLm = lms[i]; }
   109→      else if (!left) { left = h; leftLm = lms[i]; }
   110→    }
   111→
   112→    setRightPresent(!!right?.present);
   113→    setLeftPresent(!!left?.present);
   114→
   115→    // draw skeletons
   116→    if (canvas && video) {
   117→      const ctx = canvas.getContext("2d")!;
   118→      canvas.width = video.videoWidth || 640;
   119→      canvas.height = video.videoHeight || 480;
   120→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   121→      const drawHand = (lm: Landmark[] | null, color: string) => {
   122→        if (!lm) return;
   123→        const w = canvas!.width, hh = canvas!.height;
   124→        ctx.strokeStyle = color;
   125→        ctx.lineWidth = 3;
   126→        ctx.lineCap = "round";
   127→        for (const [a, b] of HAND_CONNECTIONS) {
   128→          ctx.beginPath();
   129→          ctx.moveTo(lm[a].x * w, lm[a].y * hh);
   130→          ctx.lineTo(lm[b].x * w, lm[b].y * hh);
   131→          ctx.stroke();
   132→        }
   133→        ctx.fillStyle = "rgba(255,210,80,0.95)";
   134→        for (const p of lm) {
   135→          ctx.beginPath();
   136→          ctx.arc(p.x * w, p.y * hh, 5, 0, Math.PI * 2);
   137→          ctx.fill();
   138→        }
   139→      };
   140→      drawHand(rightLm, "rgba(0,255,140,0.9)");
   141→      drawHand(leftLm, "rgba(255,120,80,0.9)");
   142→    }
   143→
   144→    if (!eng) return;
   145→
   146→    /* ---- RIGHT HAND: tempo + melody ---- */
   147→    if (right) {
   148→      setRightGesture(right.gesture);
   149→      setRightX(right.x);
   150→      setRightY(right.y);
   151→      // Y -> tempo (top = fast). 50..180
   152→      const bpm = 50 + (1 - right.y) * 130;
   153→      eng.setTempo(bpm);
   154→      setTempo(Math.round(bpm));
   155→      // X -> melody degree (7 zones)
   156→      const n = SCALES[scaleIdx].intervals.length;
   157→      const deg = zone(right.x, n * 2);
   158→      eng.setMelodyDegree(deg);
   159→      setMelodyDegree(deg);
   160→
   161→      // gesture edges
   162→      const g = right.gesture;
   163→      const prev = prevRightGestureRef.current;
   164→      if (g !== prev) {
   165→        if (g === "fist") { eng.setMelodyEnabled(false); setMelodyOn(false); }
   166→        if (g === "open") { eng.setMelodyEnabled(true); setMelodyOn(true); }
   167→        if (g === "point") {
   168→          // accent: re-trigger melody note now
   169→          eng.setMelodyEnabled(true); setMelodyOn(true);
   170→        }
   171→        prevRightGestureRef.current = g;
   172→      }
   173→    } else {
   174→      setRightGesture("idle");
   175→    }
   176→
   177→    /* ---- LEFT HAND: dynamics + progression ---- */
   178→    if (left) {
   179→      setLeftGesture(left.gesture);
   180→      setLeftX(left.x);
   181→      setLeftY(left.y);
   182→      // Y -> dynamic level (6 zones, top = ff)
   183→      const dl = 5 - zone(left.y, 6);
   184→      eng.setDynamicLevel(dl);
   185→      setDynamicLevel(dl);
   186→      // X -> progression (4 zones)
   187→      const pi = zone(left.x, PROGRESSIONS.length);
   188→      if (pi !== progIdx) {
   189→        eng.setProgression(pi);
   190→        setProgIdx(pi);
   191→      }
   192→
   193→      // gesture edges
   194→      const g = left.gesture;
   195→      const prev = prevLeftGestureRef.current;
   196→      if (g !== prev) {
   197→        if (g === "open") {
   198→          // full ensemble
   199→          (["strings", "piano", "bass", "drums", "lead"] as LayerName[]).forEach((l) => {
   200→            eng.setLayerVolume(l, layerVols[l] || 0.6);
   201→          });
   202→          setActiveLayers((Object.keys(layerVols) as LayerName[]).filter((l) => (layerVols[l] || 0) > 0.01));
   203→        }
   204→        if (g === "fist") {
   205→          // strip to strings only
   206→          (["piano", "bass", "drums", "lead"] as LayerName[]).forEach((l) => eng.setLayerVolume(l, 0));
   207→          eng.setLayerVolume("strings", layerVols.strings || 0.6);
   208→          setActiveLayers(["strings"]);
   209→        }
   210→        if (g === "pinch") {
   211→          lockedRef.current = !lockedRef.current;
   212→          eng.setLocked(lockedRef.current);
   213→          setLocked(lockedRef.current);
   214→        }
   215→        prevLeftGestureRef.current = g;
   216→      }
   217→    } else {
   218→      setLeftGesture("idle");
   219→    }
   220→
   221→    // sync display state from engine
   222→    if (eng.state) {
   223→      setChordName(eng.state.currentChordName);
   224→      setMelodyNote(eng.state.melodyNote);
   225→      setActiveLayers([...eng.state.activeLayers]);
   226→    }
   227→  }, [scaleIdx, progIdx, layerVols]);
   228→
   229→  const tracking = useHandTracking({ onHands, numHands: 2, smoothing: 0.45 });
   230→  useEffect(() => {
   231→    videoElRef.current = tracking.videoRef.current;
   232→  });
   233→
   234→  /* -------------------------------------------------------------- */
   235→  /*  Start / stop                                                  */
   236→  /* -------------------------------------------------------------- */
   237→  const start = async () => {
   238→    setError(null);
   239→    setLoading(true);
   240→    try {
   241→      if (!engineRef.current) {
   242→        engineRef.current = new MusicEngine();
   243→        await engineRef.current.init();
   244→      }
   245→      // apply current UI state
   246→      engineRef.current.setScale(scaleIdx);
   247→      engineRef.current.setProgression(progIdx);
   248→      engineRef.current.setReverb(reverbAmt);
   249→      engineRef.current.setSwing(swingAmt);
   250→      engineRef.current.setOctave(octave);
   251→      engineRef.current.setDynamicLevel(dynamicLevel);
   252→      (Object.keys(layerVols) as LayerName[]).forEach((l) => engineRef.current!.setLayerVolume(l, layerVols[l]));
   253→      engineRef.current.start();
   254→      await tracking.start();
   255→      setRunning(true);
   256→    } catch (e) {
   257→      setError(e instanceof Error ? e.message : "Failed to start");
   258→    } finally {
   259→      setLoading(false);
   260→    }
   261→  };
   262→
   263→  const stop = () => {
   264→    tracking.stop();
   265→    setRunning(false);
   266→    engineRef.current?.stop();
   267→    setRightGesture("idle");
   268→    setLeftGesture("idle");
   269→    setRightPresent(false);
   270→    setLeftPresent(false);
   271→  };
   272→
   273→  useEffect(() => {
   274→    return () => {
   275→      engineRef.current?.dispose();
   276→      engineRef.current = null;
   277→    };
   278→  }, []);
   279→
   280→  /* ---- UI handlers ---- */
   281→  const onScale = (i: number) => { setScaleIdx(i); engineRef.current?.setScale(i); };
   282→  const onProg = (i: number) => { setProgIdx(i); engineRef.current?.setProgression(i); };
   283→  const onLayerVol = (l: LayerName, v: number) => {
   284→    setLayerVols((s) => ({ ...s, [l]: v }));
   285→    engineRef.current?.setLayerVolume(l, v);
   286→  };
   287→  const onReverb = (v: number) => { setReverbAmt(v); engineRef.current?.setReverb(v); };
   288→  const onSwing = (v: number) => { setSwingAmt(v); engineRef.current?.setSwing(v); };
   289→  const onOctave = (v: number) => { setOctave(v); engineRef.current?.setOctave(v); };
   290→  const onDrop = () => engineRef.current?.triggerDrop();
   291→  const onPanic = () => {
   292→    engineRef.current?.stop();
   293→    setTimeout(() => engineRef.current?.start(), 100);
   294→  };
   295→
   296→  const intensity = dynamicLevel / 5;
   297→
   298→  return (
   299→    <div className="relative min-h-screen pt-16 sm:pt-20 pb-12">
   300→      {/* ambient background */}
   301→      <div className="absolute inset-0 overflow-hidden pointer-events-none">
   302→        <motion.div
   303→          className="absolute top-1/4 left-1/4 h-96 w-96 rounded-full blur-3xl"
   304→          style={{ background: `radial-gradient(circle, oklch(0.7 0.2 305 / ${0.15 + intensity * 0.25}), transparent)` }}
   305→          animate={{ scale: 1 + intensity * 0.3 }}
   306→          transition={{ duration: 0.4 }}
   307→        />
   308→        <motion.div
   309→          className="absolute bottom-1/4 right-1/4 h-96 w-96 rounded-full blur-3xl"
   310→          style={{ background: `radial-gradient(circle, oklch(0.72 0.19 155 / ${0.1 + intensity * 0.2}), transparent)` }}
   311→          animate={{ scale: 1 + intensity * 0.3 }}
   312→          transition={{ duration: 0.4 }}
   313→        />
   314→      </div>
   315→
   316→      <div className="relative mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   317→        {/* header */}
   318→        <div className="mb-8 text-center">
   319→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="inline-flex items-center gap-2 px-3 py-1 rounded-full glass text-xs font-medium text-chart-4 mb-4">
   320→            <Music className="h-3.5 w-3.5" /> Two-handed conducting · Generative · Tone.js
   321→          </motion.div>
   322→          <h1 className="text-4xl sm:text-5xl font-bold tracking-tight">
   323→            AI Conducting <span className="text-gradient">Orchestra</span>
   324→          </h1>
   325→          <p className="mt-3 text-muted-foreground max-w-2xl mx-auto">
   326→            Conduct a live, procedurally-generated cinematic score with <span className="text-primary">both hands</span>.
   327→            Right hand sets tempo &amp; melody; left hand sets dynamics &amp; harmony. Every note is synthesized in real time.
   328→          </p>
   329→        </div>
   330→
   331→        <div className="grid lg:grid-cols-[1.3fr_1fr] gap-6">
   332→          {/* ---- camera + overlays ---- */}
   333→          <div className="flex flex-col gap-4">
   334→            <div className="relative rounded-2xl overflow-hidden glass-strong aspect-video bg-black">
   335→              <video ref={tracking.videoRef} playsInline muted className="absolute inset-0 h-full w-full object-cover opacity-0" />
   336→              <canvas ref={canvasRef} className="absolute inset-0 h-full w-full object-cover" />
   337→
   338→              {!running && (
   339→                <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 grid-bg">
   340→                  <div className="relative">
   341→                    <div className="absolute inset-0 bg-chart-4/30 blur-2xl rounded-full animate-pulse-glow" />
   342→                    <Music className="relative h-16 w-16 text-chart-4/70" />
   343→                  </div>
   344→                  <p className="text-muted-foreground text-sm max-w-xs text-center">
   345→                    {loading ? "Loading model + audio engine…" : "Click start to begin conducting"}
   346→                  </p>
   347→                  {loading && <Loader2 className="h-5 w-5 animate-spin text-chart-4" />}
   348→                </div>
   349→              )}
   350→
   351→              <AnimatePresence>
   352→                {running && (
   353→                  <>
   354→                    {/* right hand badge */}
   355→                    <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="absolute top-3 left-3 flex items-center gap-2 px-3 py-1.5 rounded-lg glass-strong">
   356→                      <span className={cn("h-2 w-2 rounded-full", rightPresent ? "bg-primary animate-pulse" : "bg-muted-foreground")} />
   357→                      <span className="text-xs font-medium">R: <span className="capitalize text-primary">{rightGesture}</span></span>
   358→                    </motion.div>
   359→                    {/* left hand badge */}
   360→                    <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="absolute top-3 right-3 flex items-center gap-2 px-3 py-1.5 rounded-lg glass-strong">
   361→                      <span className={cn("h-2 w-2 rounded-full", leftPresent ? "bg-chart-3 animate-pulse" : "bg-muted-foreground")} />
   362→                      <span className="text-xs font-medium">L: <span className="capitalize text-chart-3">{leftGesture}</span></span>
   363→                    </motion.div>
   364→                    {/* chord + melody */}
   365→                    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="absolute bottom-3 left-3 flex gap-2">
   366→                      <div className="px-3 py-1.5 rounded-lg glass-strong text-center">
   367→                        <div className="text-[9px] text-muted-foreground uppercase">Chord</div>
   368→                        <div className="text-lg font-bold text-chart-3 leading-none">{chordName}</div>
   369→                      </div>
   370→                      <div className="px-3 py-1.5 rounded-lg glass-strong text-center">
   371→                        <div className="text-[9px] text-muted-foreground uppercase">Melody</div>
   372→                        <div className="text-lg font-bold text-chart-4 leading-none">{melodyNote}</div>
   373→                      </div>
   374→                    </motion.div>
   375→                    {locked && (
   376→                      <motion.div initial={{ opacity: 0, scale: 0.8 }} animate={{ opacity: 1, scale: 1 }} className="absolute bottom-3 right-3 flex items-center gap-1.5 px-3 py-1.5 rounded-lg glass-strong">
   377→                        <Lock className="h-3.5 w-3.5 text-chart-3" />
   378→                        <span className="text-xs">Locked</span>
   379→                      </motion.div>
   380→                    )}
   381→                  </>
   382→                )}
   383→              </AnimatePresence>
   384→
   385→              {error && (
   386→                <div className="absolute bottom-3 left-3 right-3 px-4 py-3 rounded-lg bg-destructive/20 border border-destructive/40 text-sm text-destructive">{error}</div>
   387→              )}
   388→            </div>
   389→
   390→            {/* start/stop + drop + panic */}
   391→            <div className="grid grid-cols-3 gap-3">
   392→              {!running ? (
   393→                <button onClick={start} disabled={loading} className="col-span-3 flex items-center justify-center gap-2 px-4 py-3.5 rounded-xl font-medium hover:brightness-110 transition-all hover:glow-cyan disabled:opacity-50 text-white" style={{ backgroundColor: "oklch(0.62 0.22 305)" }}>
   394→                  {loading ? <Loader2 className="h-5 w-5 animate-spin" /> : <Play className="h-5 w-5" />}
   395→                  {loading ? "Loading…" : "Start Conducting"}
   396→                </button>
   397→              ) : (
   398→                <>
   399→                  <button onClick={stop} className="col-span-1 flex items-center justify-center gap-2 px-4 py-3.5 rounded-xl glass text-foreground font-medium hover:bg-white/10 transition-all">
   400→                    <Pause className="h-5 w-5" /> Stop
   401→                  </button>
   402→                  <button onClick={onDrop} className="col-span-1 flex items-center justify-center gap-2 px-4 py-3.5 rounded-xl bg-chart-4 text-white font-medium hover:brightness-110 transition-all" style={{ backgroundColor: "oklch(0.62 0.22 305)" }}>
   403→                    <Zap className="h-5 w-5" /> Drop
   404→                  </button>
   405→                  <button onClick={onPanic} className="col-span-1 flex items-center justify-center gap-2 px-4 py-3.5 rounded-xl glass text-foreground font-medium hover:bg-white/10 transition-all">
   406→                    <RotateCcw className="h-5 w-5" /> Reset
   407→                  </button>
   408→                </>
   409→              )}
   410→            </div>
   411→
   412→            {/* hand role guide */}
   413→            <div className="grid sm:grid-cols-2 gap-3">
   414→              <div className="rounded-xl glass p-4">
   415→                <div className="text-xs font-medium text-primary uppercase tracking-wider mb-2">Right Hand ✋</div>
   416→                <ul className="text-xs text-muted-foreground space-y-1">
   417→                  <li>↕ <span className="text-foreground">Height</span> → Tempo (50–180)</li>
   418→                  <li>↔ <span className="text-foreground">X position</span> → Melody note</li>
   419→                  <li>✊ <span className="text-foreground">Fist</span> → Mute melody</li>
   420→                  <li>🖐 <span className="text-foreground">Open</span> → Melody on</li>
   421→                </ul>
   422→              </div>
   423→              <div className="rounded-xl glass p-4">
   424→                <div className="text-xs font-medium text-chart-3 uppercase tracking-wider mb-2">Left Hand ✋</div>
   425→                <ul className="text-xs text-muted-foreground space-y-1">
   426→                  <li>↕ <span className="text-foreground">Height</span> → Dynamics (pp–ff)</li>
   427→                  <li>↔ <span className="text-foreground">X position</span> → Progression</li>
   428→                  <li>✊ <span className="text-foreground">Fist</span> → Strings only</li>
   429→                  <li>🤏 <span className="text-foreground">Pinch</span> → Lock chord</li>
   430→                </ul>
   431→              </div>
   432→            </div>
   433→          </div>

   548→        </div>
